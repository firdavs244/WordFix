    # Media
    image_url: str                  # Rasm URL
    audio_url: str                  # Audio URL (TTS orqali yaratiladi)

    # Foydalanuvchi kiritgan
    notes: str                      # Eslatmalar
    mnemonic: str                   # Mnemonik (AI to'ldiradi)
    usage_notes: str                # Ishlatish bo'yicha izohlar (AI to'ldiradi)

    # Kategoriya
    category_id: UUID | None        # Kategoriya ID (ixtiyoriy)
    category: WordCategoryEntity | None  # Nested category object
    tags: list[str]                 # Teglar

    # Qiyinlik
    difficulty_level: str           # "easy" | "medium" | "hard"

    # AI Enrichment holati
    is_enriched: bool               # Boyitilganmi?
    enrichment_status: str          # "pending" | "enriching" | "enriched" | "failed"
    enrichment_error: str           # Xato xabari (agar failed bo'lsa)
    enriched_at: datetime | None    # Boyitilgan vaqt

    # Review statistikasi
    confidence_score: float         # Ishonch darajasi (0-100)
    next_review_at: datetime | None # Keyingi review vaqti (SM-2)
    review_count: int               # Jami review soni
    correct_count: int              # To'g'ri javoblar
    incorrect_count: int            # Noto'g'ri javoblar
    last_reviewed_at: datetime | None # Oxirgi review vaqti
    is_mastered: bool               # O'zlashtirilganmi? (confidence >= 95 va rep >= 5)

    # SM-2 Spaced Repetition maydonlari
    easiness_factor: float          # EF (default: 2.5, min: 1.3)
    repetition_number: int          # Ketma-ket to'g'ri javoblar soni
    interval_days: int              # Keyingi review'gacha kunlar

    # Arxiv
    is_archived: bool               # Arxivlanganmi?
    archived_at: datetime | None    # Arxivlangan vaqt
    is_active: bool                 # Faolmi?

    # Timestamps
    created_at: datetime | None
    updated_at: datetime | None

    # Hisoblangan property
    @property
    def accuracy_rate(self) -> float:
        """Aniqlik darajasi: correct_count / review_count * 100"""
        if self.review_count == 0:
            return 0.0
        return round((self.correct_count / self.review_count) * 100, 1)

    def validate(self) -> None:
        """Validatsiya: so'z bo'sh emas, difficulty to'g'ri, confidence 0-100"""

5. SM-2 SPACED REPETITION ALGORITM
================================================================================

Fayl: apps/words/domain/services/review_service.py
Klass: SpacedRepetitionService

SM-2 algoritmi - so'zlarni samarali takrorlash uchun ilmiy asoslangan tizim.
Har bir review'dan keyin keyingi review vaqtini hisoblaydi.

QUALITY SCALE (0-5):
  0 = To'liq unutilgan (Complete blackout)
  1 = Noto'g'ri, lekin javobni ko'rganda esladi
  2 = Noto'g'ri, lekin javob oson tuyuldi
  3 = To'g'ri, lekin katta qiyinchilik bilan
  4 = To'g'ri, biroz ikkilanish bilan
  5 = Mukammal javob

ALGORITM ISHLASHI:

def calculate_next_review(word_entity, quality):
    ef = word_entity.easiness_factor    # Osonlik koeffitsienti (default 2.5)
    rep = word_entity.repetition_number  # Ketma-ket to'g'ri javoblar
    interval = word_entity.interval_days # Hozirgi interval

    if quality >= 3:  # To'g'ri javob
        if rep == 0:
            interval = 1    # 1 kun
        elif rep == 1:
            interval = 6    # 6 kun
        else:
            interval = round(interval * ef)  # interval * EF
        rep += 1
    else:  # Noto'g'ri javob
        rep = 0
        interval = 1    # Qaytadan 1 kundan boshlash

    # EF yangilash (minimum 1.3)
    ef = ef + (0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02))
    if ef < 1.3:
        ef = 1.3

    # Natijalar
    word.easiness_factor = ef
    word.repetition_number = rep
    word.interval_days = interval
    word.next_review_at = now + timedelta(days=interval)

    # Confidence hisoblash
    word.confidence_score = calculate_confidence(word)

    # Mastered tekshirish
    if confidence >= 95 and rep >= 5:
        word.is_mastered = True

CONFIDENCE HISOBLASH:
    base = (correct_count / review_count) * 100
    + recency_bonus (0-10): oxirgi reviewdan beri kunlar
    + streak_bonus (0-20): repetition_number * 5
    = min(base + recency + streak, 100.0)

INTERVAL MISOLLARI:
    quality=5, rep=0 → 1 kun
    quality=5, rep=1 → 6 kun
    quality=5, rep=2 → 6 * 2.6 = 16 kun
    quality=5, rep=3 → 16 * 2.7 = 43 kun
    quality=5, rep=4 → 43 * 2.8 = 120 kun
    quality=2 (noto'g'ri) → 1 kun (qaytadan boshlash)
